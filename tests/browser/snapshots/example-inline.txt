// A first look at the wages that ship with Zeff: a summary, a regression and two figures.
// Press Run to see each result under the line that made it.

let wages = dataset(wages)
let w2010 = wages.where(year == 2010)

// Describe: hourly wage, schooling, experience and hours, for men and for women.
w2010.summary([wage, educ, exper, hours], by: female)
